2.2 确认对话框：替掉 4 处 window.confirm（对应 #9、#10）

organic.css 里 .dialog / .dialog-backdrop 已经有样式，一次没用过。一个 hook 加一个组件，替掉 VehiclesPanel.tsx:180,292,305 和 AccountsPage.tsx:93，同时把漏掉确认的「恢复为草稿」「停用账号」补上。

// shared 或 admin-web/src/ui/useConfirm.tsx（新文件）
import { useCallback, useEffect, useRef, useState } from "react";

export type ConfirmSpec = {
  title: string;          // 问句
  body: string;           // 后果 + 可逆性
  confirmLabel: string;   // 复述动作，不写「确定」
  danger?: boolean;
};

export function useConfirm() {
  const [spec, setSpec] = useState<ConfirmSpec | null>(null);
  const [busy, setBusy] = useState(false);
  const resolver = useRef<((ok: boolean) => void) | null>(null);

  const confirm = useCallback((next: ConfirmSpec) => {
    setSpec(next);
    return new Promise<boolean>((resolve) => { resolver.current = resolve; });
  }, []);

  function close(ok: boolean) {
    resolver.current?.(ok);
    resolver.current = null;
    setSpec(null);
    setBusy(false);
  }

  const dialog = spec ? (
    <ConfirmDialog
      spec={spec}
      busy={busy}
      onCancel={() => close(false)}
      onConfirm={() => { setBusy(true); close(true); }}
    />
  ) : null;

  return { confirm, dialog };
}

function ConfirmDialog({ spec, busy, onCancel, onConfirm }: {
  spec: ConfirmSpec; busy: boolean; onCancel: () => void; onConfirm: () => void;
}) {
  const cancelRef = useRef<HTMLButtonElement>(null);

  useEffect(() => {
    cancelRef.current?.focus();
    const onKey = (e: KeyboardEvent) => { if (e.key === "Escape") onCancel(); };
    document.addEventListener("keydown", onKey);
    return () => document.removeEventListener("keydown", onKey);
  }, [onCancel]);

  return (
    <div className="dialog-backdrop" onClick={onCancel}>
      <div
        className="dialog"
        role="dialog"
        aria-modal="true"
        aria-labelledby="confirm-title"
        onClick={(e) => e.stopPropagation()}
      >
        <div className="dialog-title" id="confirm-title">{spec.title}</div>
        <div className="dialog-body">{spec.body}</div>
        <div className="dialog-actions">
          <button ref={cancelRef} type="button" className="btn btn-ghost"
                  onClick={onCancel} disabled={busy}>取消</button>
          <button type="button" className="btn btn-primary"
                  onClick={onConfirm} disabled={busy}>
            {busy ? "处理中…" : spec.confirmLabel}
          </button>
        </div>
      </div>
    </div>
  );
}

2.3 字段级校验（对应 #14、#15）

一个 Field 包装 + 一张纯函数校验表。校验在 onBlur 触发（不是每次按键），提交时全量再跑一遍。服务端仍是最终权威，前端只负责别让明显错的值发出去。

// admin-web/src/vehicles/validate.ts（新文件）
const THIS_YEAR = new Date().getFullYear();

export const RULES = {
  brand:  (v: string) => (v.trim() ? null : "请填品牌"),
  model:  (v: string) => (v.trim() ? null : "请填车型"),
  registrationYear: (v: number) =>
    v >= 1980 && v <= THIS_YEAR ? null : `请填 1980 至 ${THIS_YEAR} 之间的年份`,
  mileageKm: (v: number) =>
    v >= 0 && v <= 2_000_000 ? null : "里程应在 0 到 200 万公里之间",
  transferCount: (v: number) =>
    v >= 0 && v <= 20 ? null : "过户次数应在 0 到 20 之间",
  displacementL: (v: string) =>
    !v || (Number(v) > 0 && Number(v) <= 10) ? null : "排量应在 0 到 10 升之间",
} as const;

export type FieldErrors = Partial<Record<keyof typeof RULES, string>>;
// admin-web/src/ui/Field.tsx（新文件）
import type { ReactNode } from "react";

const REQUIRED_MARK = { color: "var(--color-danger)" };

export function Field({ label, error, required, hint, children }: {
  label: string;
  error?: string;
  required?: boolean;
  hint?: string;
  children: ReactNode;
}) {
  return (
    <label className={error ? "field field-error" : "field"}>
      <span>
        {label}
        {required ? <span aria-hidden="true" style={REQUIRED_MARK}> *</span> : null}
      </span>
      {children}
      {error
        ? <span className="field-error-text" role="alert">{error}</span>
        : hint
          ? <span className="field-hint">{hint}</span>
          : null}
    </label>
  );
}

3.3 后台车辆列表：去掉 N+1（对应 #22、#23、#4）

首选方案是契约层加字段，一次请求拿全。若这一版不动契约，用下面的兜底改法：请求数不变，但收敛成一次 setState，卡片不再逐张闪动；切页时先清空，避免旧封面错配。

//   coverUrl: string | null
//   price: { type, amount } | null      ← 顺带解掉 #4，卡片能显示价格
// 前端随即删掉整个 covers 状态与那段 Promise.all，请求 21 → 1。

// 兜底（不动契约）：VehiclesPanel.tsx:57-68 替换
setCovers({});                                    // 切页先清，避免错配（#23）

const entries = await Promise.all(
  data.items.map(async (v) => {
    try {
      const imgs = await api<{ items: ImageItem[] }>(
        `/admin/vehicles/${v.id}/images`,
      );
      return [v.id, imgs.items[0]?.url] as const;
    } catch {
      return [v.id, undefined] as const;
    }
  }),
);

// 一次 setState，取代原来逐条 setCovers 的 20 次
setCovers(Object.fromEntries(entries.filter(([, url]) => url)));

权衡：兜底方案把「封面逐张出现」换成「等全部到齐再一起出现」。20 个并发请求里最慢的那个决定了封面出现的时间，所以骨架屏会多停留一会儿。这是划算的 —— 持续闪动比多等 200ms 更伤。真要根治只能走契约层。
